class Rohling
{

    public function schilde()
    {
		if($this->energie<=0) return 5;
		if($this->schilde==0) return 6;
		if($this->dock>0 && $this->schildstatus==0) $this->dock=0;
		$this->schildstatus=$this->schildstatus==0?1:0;
		$this->energie--;
		mysql_query("UPDATE schiffe SET dock='$this->dock',energie=energie-1,schildstatus='$this->schildstatus' WHERE id='$this->id'");
		return 0;
    }
	
	public function schildaufladen($anzahl)
	{
		if($anzahl+$this->schilde>$this->maxschilde) $anzahl=$this->maxschilde-$this->schilde;
		if($anzahl<0) $anzahl=0;
		if($this->energie<$anzahl) return 7;
		if($this->schildstatus==1) return 8;
		$this->energie-=$anzahl;
		$this->schilde+=$anzahl;
		mysql_query("UPDATE schiffe SET energie='$this->energie',schilde='$this->schilde' WHERE id='$this->id'");
