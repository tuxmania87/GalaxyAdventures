class LISTE is
readonly head:INT;
readonly tail:LISTE;
create:LISTE is 
res:=#LISTE; end;
cons(a:INT):LISTE is
l:LISTE:=copy;
head:=a;
tail:=l;
res:=self;
end;
isEmpty:BOOL is
res:=tail=void;
end;

foldr(f:procedure(INT,INT):INT;z:INT) is
if isEmpty=true then <<z<<"\n"; return; end;
z:=f(head,z);
tail.foldr(f,z);
end;

mal(a,b:INT):INT is
res:=a*b;
end;

produkt is
self.foldr(bind mal(,),1);
end;
end;
